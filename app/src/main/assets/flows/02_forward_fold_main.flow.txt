# YogaFlow Live Coach Flow
# Flow ID: 02_forward_fold_main
# Pose: Forward Fold
# Role: Main Stretch
# Version: 1.0
# Source intent: YouTube-aligned live coach sequence
# Safety rule: guide from hip hinge, keep knees long but not locked, avoid forcing depth.

[FLOW]
id = 02_forward_fold_main
name = 兩腿水平拉筋 · Forward Fold
pose = forward_fold
language = zh-TW
level = beginner

[STEP 1]
state = SETUP
duration_ms = 4000
cue = 先坐穩或站穩，雙腿打開，膝蓋伸直但不要鎖死，腳尖自然朝上。
detect = ready_forward_fold
correction = 先把雙腿打開，膝蓋伸長。如果膝蓋彎曲，先不要急著前彎。

[STEP 2]
state = SETUP
duration_ms = 3000
cue = 吸氣，把背拉長，胸口打開，雙手可以放在地板或腿上支撐。
detect = tall_spine_setup
correction = 先不要低頭壓下去，胸口打開，讓背變長。

[STEP 3]
state = MOVEMENT
duration_ms = 5000
cue = 吐氣，從髖部慢慢往前傾，不是用脖子或上背硬壓。
detect = hip_hinge
correction = 你現在有點用背彎，先停一下，背打直，再從髖部往前。

[STEP 4]
state = MOVEMENT
duration_ms = 5000
cue = 維持膝蓋伸長，身體只到舒服的位置，雙手往前或扶住腿都可以。
detect = controlled_forward_fold
correction = 如果膝蓋彎了，先減少前傾深度，把腿重新伸長。

[STEP 5]
state = HOLD
duration_ms = 8000
cue = 停在這裡，肩膀放鬆，呼吸保持穩定，不要憋氣。
detect = forward_hold
correction = 不需要再壓更深，先穩住呼吸，讓背和大腿慢慢放鬆。

[STEP 6]
state = HOLD
duration_ms = 8000
cue = 再維持一下。吸氣感覺背拉長，吐氣讓身體自然沉一點點。
detect = forward_hold
correction = 如果感覺拉扯太強，身體退回一點，保持安全伸展。

[STEP 7]
state = TRANSITION
duration_ms = 5000
cue = 準備回來。吸氣，雙手支撐，從髖部帶動身體慢慢起來。
detect = return_standing
correction = 不要突然抬頭或快速起身，慢慢回到中間。

[STEP 8]
state = HOLD
duration_ms = 3000
cue = 回到中間，肩膀放鬆，感覺大腿後側和背部的伸展。
detect = neutral_finish
correction = 如果還覺得緊，保持腿伸長，先穩定呼吸。

[END]
cue = 前彎完成，很好。這一段重點是腿伸長、背拉長、從髖部前傾。
