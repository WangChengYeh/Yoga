[STEP 2]

[STEP 3]
state = HOLD
duration_ms = 8000
cue = 停在扭轉位置，感覺脊椎延伸。
detect = twist_hold
correction = 如果聳肩，提醒放鬆肩膀。

[STEP 4]
state = TRANSITION
duration_ms = 4000
cue = 慢慢回到中間，換邊或放鬆。
detect = return_center
correction = 如果動作太快，提醒慢一點。

[END]
cue = 收操完成，身體恢復平衡。
